def register(session: Session, payload: dict) -> dict:
    validated = validate_registration(payload)
    user = User(email=validated["email"].strip().casefold(),
                password_hash=hash_password(validated["password"]), name=validated["name"])
    try:
        session.add(user)
        session.commit()  # Persist the user and generate its UUID before returning.
    except IntegrityError as error:
        session.rollback()
        existing = get_user_by_email(session, user.email)
        session.rollback()
        if existing is not None:
            raise HTTPException(409, "Email already registered") from error
        raise
    except SQLAlchemyError:
        session.rollback()
        raise
    return public_user(user)
loginaccounts/handlers.py
def login(session: Session, payload: dict, redis_client=None) -> dict:
    validated = LoginInput.model_validate(payload)
    email = str(validated.email).strip().casefold()
    if redis_client is not None:
        check_login_limit(redis_client, email)
    user = get_user_by_email(session, email)
    if user is None or not verify_password(validated.password, user.password_hash):
        raise HTTPException(401, "Invalid email or password", headers={"WWW-Authenticate": "Bearer"})
    token = issue_token(str(user.id), jwt_secret(), int(time.time()))
    return {"access_token": token, "token_type": "bearer"}

def bearer_token(authorization: str) -> str:
    parts = authorization.split() if isinstance(authorization, str) else []
    if len(parts) != 2 or parts[0].casefold() != "bearer":
        raise HTTPException(401, "Invalid credentials", headers={"WWW-Authenticate": "Bearer"})
    return parts[1]
profileaccounts/handlers.py
def profile(session: Session, authorization: str, redis_client=None) -> dict:
    token = bearer_token(authorization)
    secret = jwt_secret()
    try:
        user_id = UUID(decode_token(token, secret, int(time.time())))
    except ValueError as error:
        raise HTTPException(401, "Invalid credentials", headers={"WWW-Authenticate": "Bearer"}) from error
    if redis_client is not None and is_revoked(redis_client, token):
        raise HTTPException(401, "Invalid credentials", headers={"WWW-Authenticate": "Bearer"})
    user = session.get(User, user_id)
    if user is None:
        raise HTTPException(401, "Invalid credentials", headers={"WWW-Authenticate": "Bearer"})
    return public_user(user)

def issue_token(user_id: str, secret: str, now: int, ttl: int = 900) -> str:
    payload = {
        "sub": user_id,
        "iat": now,
        "exp": now + ttl,
    }
    token = jwt.encode(payload, secret, algorithm="HS256")
    return token
decode_tokencore/tokens.py
def decode_token(token: str, secret: str, now: int) -> str:
    try:
        decoded = jwt.decode(
            token,
            secret,
            algorithms=["HS256"],
            options={
                "require": ["sub", "iat", "exp"],
                # Check time below with `now`, keeping boundary tests deterministic.
                # These options do not disable signature verification.
                "verify_exp": False,
                "verify_iat": False,
            },
        )
    except jwt.PyJWTError as error:
        raise ValueError("Invalid token") from error

    if not isinstance(decoded["sub"], str) or not decoded["sub"]:
        raise ValueError("Invalid subject claim")
    # bool is an int subclass in Python, but is not a valid timestamp here.
    if type(decoded["iat"]) is not int:
        raise ValueError("Invalid iat claim")
    if type(decoded["exp"]) is not int:
        raise ValueError("Invalid expiration claim")
    if decoded["iat"] > now:
        raise ValueError("Token issued in the future")
    if now >= decoded["exp"]:
        raise ValueError("Token has expired")
    return decoded["sub"]

check_login_limitcore/redis_client.py
def check_login_limit(client, email: str) -> None:
    key = "auth:login:" + sha256(email.encode()).hexdigest()
    try:
        # Atomic increment + expiring window; expiry is set only on first attempt.
        count = client.eval("local n=redis.call('INCR',KEYS[1]); "
                            "if n==1 then redis.call('EXPIRE',KEYS[1],ARGV[1]) end; return n",
                            1, key, 60)
    except RedisError as error:
        raise HTTPException(503, "Authentication storage unavailable") from error
    if count > 20:
        raise HTTPException(429, "Too many login attempts", headers={"Retry-After": "60"})
is_revokedcore/redis_client.py
def is_revoked(client, token: str) -> bool:
    try:
        return bool(client.exists(token_key(token)))
    except RedisError as error:
        # Never silently bypass revocation when its backing store is unavailable.
        raise HTTPException(503, "Authentication storage unavailable") from error
revoke_tokencore/redis_client.py
def revoke_token(client, token: str, secret: str) -> None:
    try:
        payload = jwt.decode(token, secret, algorithms=["HS256"])
    except jwt.PyJWTError as error:
        raise HTTPException(401, "Invalid credentials") from error
    ttl = max(1, payload["exp"] - int(time()))
    try:
        client.setex(token_key(token), ttl, "1")
    except RedisError as error:
        raise HTTPException(503, "Authentication storage unavailable") from error

set_quantitycart/service.py
def set_quantity(session: Session, user_id: str, product_id: str, quantity: int) -> dict:
    if type(quantity) is not int or not 0 <= quantity <= 100:
        raise ValueError("Quantity must be between 0 and 100")
    owner, item_id = UUID(user_id), UUID(product_id)
    try:
        # Lock the owner to serialize concurrent cart creation/updates per user.
        if session.scalar(select(User).where(User.id == owner).with_for_update()) is None:
            raise HTTPException(404, "User not found")
        product = session.get(Product, item_id)
        if product is None:
            raise HTTPException(404, "Product not found")
        inventory = session.scalar(select(Inventory).where(Inventory.product_id == item_id))
        if quantity and (inventory is None or quantity > inventory.quantity):
            raise HTTPException(409, "Requested quantity exceeds stock")
        cart = session.scalar(select(Cart).where(Cart.user_id == owner))
        if cart is None and quantity:
            cart = Cart(user_id=owner)
            session.add(cart)
            session.flush()
        line = session.scalar(select(CartItem).where(CartItem.cart_id == cart.id,
                              CartItem.product_id == item_id)) if cart else None
        if quantity == 0:
            if line is not None:
                session.delete(line)
        elif line is None:
            session.add(CartItem(cart_id=cart.id, product_id=item_id, quantity=quantity))
        else:
            line.quantity = quantity
        session.commit()  # A cart update never decrements inventory.
    except Exception:
        session.rollback()
        raise
    return {"product_id": product_id, "quantity": quantity}

def request_fingerprint(request: CreateOrderRequest, scope: str) -> str:
    payload = request.model_dump(mode="json")
    payload["lines"].sort(key=lambda line: line["productId"])
    payload["checkout"]["email"] = payload["checkout"]["email"].strip().casefold()
    return sha256(json.dumps({"scope": scope, "request": payload}, sort_keys=True,
                             separators=(",", ":")).encode()).hexdigest()

def replay_order(order: Order, fingerprint: str, scope: str) -> OrderReceipt:
    if order.owner_scope != scope or order.request_fingerprint != fingerprint:
        # Legacy orders with no fingerprint cannot establish a safe replay.
        raise IdempotencyConflictError("Idempotency key already used for another checkout")

create_orderorders/service.py
def create_order(session: Session, request: CreateOrderRequest, idempotency_key: str,
                 user_id: str | None = None, anonymous_id: str | None = None) -> OrderReceipt:
    owner = UUID(user_id) if user_id else None
    anonymous = UUID(anonymous_id) if anonymous_id else None
    scope = f"user:{owner}" if owner else f"guest:{anonymous or 'legacy-direct-call'}"
    fingerprint = request_fingerprint(request, scope)
    ids = [line.productId for line in request.lines]
    if len(ids) != len(set(ids)):
        raise DuplicateProductError("Each product may appear only once")
    try:
        existing = session.scalar(select(Order).where(Order.idempotency_key == idempotency_key))
        if existing:
            result = replay_order(existing, fingerprint, scope)
            session.rollback()  # End the read-only retry transaction.
            return result
        rows = session.execute(select(Product, Inventory).join(Inventory, Inventory.product_id == Product.id)
                               .where(Product.id.in_(ids)).order_by(Product.id).with_for_update(of=Inventory)
                               .execution_options(populate_existing=True)).all()
        # A matching retry may have waited on the first request's inventory lock.
        existing = session.scalar(select(Order).where(Order.idempotency_key == idempotency_key))
        if existing:
            result = replay_order(existing, fingerprint, scope)
            session.rollback()
            return result
        products = {product.id: (product, stock) for product, stock in rows}
        if len(products) != len(ids):
            raise ProductNotFoundError("One or more products do not exist")
        if any(products[line.productId][1].quantity < line.quantity for line in request.lines):
            raise InsufficientInventoryError("Not enough inventory")
        subtotal = sum((products[line.productId][0].price * line.quantity for line in request.lines), Decimal("0.00"))
        shipping = Decimal("0.00") if subtotal >= Decimal("150.00") else Decimal("12.00")
        order = Order(status=OrderStatus.CONFIRMED, idempotency_key=idempotency_key,
                      request_fingerprint=fingerprint, owner_scope=scope, user_id=owner, anonymous_id=anonymous,
                      customer_email=str(request.checkout.email), customer_name=request.checkout.fullName,
                      address=request.checkout.address, city=request.checkout.city,
                      postal_code=request.checkout.postalCode, country=request.checkout.country,
                      subtotal=subtotal, shipping=shipping, tax=Decimal("0"), discount=Decimal("0"),
                      total=subtotal + shipping, currency="SGD")
        session.add(order)
        session.flush()  # Claim the unique idempotency key before changing stock.
        for line in request.lines:
            product, stock = products[line.productId]
            # Conditional SQL decrement also guards against concurrent last-unit sales.
            changed = session.execute(update(Inventory).where(Inventory.product_id == product.id,
                                      Inventory.quantity >= line.quantity)
                                      .values(quantity=Inventory.quantity - line.quantity)
                                      .execution_options(synchronize_session="fetch"))
            if changed.rowcount != 1:
                raise InsufficientInventoryError("Not enough inventory")
            order.lines.append(OrderLine(product_id=product.id, product_slug=product.slug,
                               product_name=product.name, quantity=line.quantity, unit_price=product.price,
                               line_total=product.price * line.quantity, tax=Decimal("0"),
                               discount=Decimal("0"), currency=product.currency))
            # Purchases are recorded only inside a successful checkout transaction.
            session.add(Interaction(event_id=uuid5(NAMESPACE_URL, f"purchase:{order.id}:{product.id}"),
                                    kind="purchase", product_id=product.id, user_id=owner,
                                    anonymous_id=anonymous or uuid4()))
        if owner:
            cart_id = session.scalar(select(Cart.id).where(Cart.user_id == owner))
            if cart_id:
                session.execute(delete(CartItem).where(CartItem.cart_id == cart_id, CartItem.product_id.in_(ids)))
        session.flush()
        result = _to_order_receipt(order)
        session.commit()  # Order, inventory, cart cleanup and purchase events are atomic.
        return result
    except IntegrityError as error:
        session.rollback()
        # A concurrent request may have claimed the same key while this one waited.
        existing = session.scalar(select(Order).where(Order.idempotency_key == idempotency_key))
        if existing is None:
            session.rollback()
            raise
        try:
            return replay_order(existing, fingerprint, scope)
        finally:
            session.rollback()
    except Exception:
        session.rollback()
        raise

record_eventinteractions/service.py
def record_event(session: Session, event: dict, user_id: str | None, anonymous_id: str) -> dict:
    parsed = EventInput.model_validate(event)
    owner = UUID(user_id) if user_id else None
    anonymous = UUID(anonymous_id)
    def replay(existing):
        if (existing.user_id, existing.anonymous_id, existing.kind, existing.product_id, existing.query) != (
                owner, anonymous, parsed.kind, parsed.product_id, parsed.query):
            raise HTTPException(409, "Event ID already used")
        return {"event_id": str(parsed.event_id), "accepted": True}
    try:
        existing = session.scalar(select(Interaction).where(Interaction.event_id == parsed.event_id))
        if existing:
            return replay(existing)
        if parsed.product_id and session.get(Product, parsed.product_id) is None:
            raise HTTPException(404, "Product not found")
        session.add(Interaction(event_id=parsed.event_id, kind=parsed.kind, product_id=parsed.product_id,
                                user_id=owner, anonymous_id=anonymous, query=parsed.query))
        session.commit()
    except IntegrityError:
        session.rollback()
        existing = session.scalar(select(Interaction).where(Interaction.event_id == parsed.event_id))
        if existing is None:
            raise
        return replay(existing)
    except Exception:
        session.rollback()
        raise
    return {"event_id": str(parsed.event_id), "accepted": True}

list_productscatalog/service.py
def list_products(
    session: Session,
    category: str | None = None,
    search: str | None = None,
    limit: int = 10,
    offset: int = 0,
) -> list[ProductResponse]:
    if not 1 <= limit <= 100 or offset < 0:
        raise ValueError("Invalid pagination")
    if not category and not search:
        from app.catalog.performance import paginated_catalog
        return paginated_catalog(session, limit, offset)
    statement = select(Product).options(
        selectinload(Product.category),
        selectinload(Product.inventory_record),
    )
    if category:
        statement = statement.join(Product.category).where(Category.name == category)
    if search:
        statement = statement.where(
            or_(
                Product.name.ilike(f"%{search}%"),
                Product.description.ilike(f"%{search}%"),
            )
        )
    statement = statement.order_by(Product.name, Product.id).offset(offset).limit(limit)
    products = session.scalars(statement).all()
    responses: list[ProductResponse] = []

    for product in products:
        response = _to_product_response(product)
        responses.append(response)

    return responses

                    Interaction.product_id.is_not(None)).order_by(Interaction.created_at.desc()).limit(30)).all()
        seeds.update(history)
        purchased.update(session.scalars(select(OrderLine.product_id).join(Order)
                         .where(Order.user_id == owner, Order.status == OrderStatus.CONFIRMED)))
    seed_categories = {product.category_id for product in products if product.id in seeds}
    counts = Counter()
    if mode == "frequently_bought" and seed_product_ids:
        baskets = session.scalars(select(OrderLine.order_id).join(Order).where(
            OrderLine.product_id == seed_product_ids[0], Order.status == OrderStatus.CONFIRMED)).all()
        if baskets:
            # Count each companion once per completed order, never text similarity.
            pairs = session.execute(select(OrderLine.order_id, OrderLine.product_id)
                                    .where(OrderLine.order_id.in_(baskets)).distinct())
            counts.update(product_id for _, product_id in pairs if product_id not in seeds)
    candidates = [p for p in available.values() if p.id not in seeds and p.id not in purchased]
    strategy = "featured_fallback"
    reason = "Featured catalogue selection; learning models are not trained yet"
    if mode == "frequently_bought":
        candidates = [p for p in candidates if counts[p.id] > 0]
        candidates.sort(key=lambda p: (-counts[p.id], p.name, str(p.id)))
        strategy, reason = "order_cooccurrence", "Bought in the same completed orders"
    elif seed_categories and mode != "trending":
        candidates.sort(key=lambda p: (p.category_id not in seed_categories, not p.featured, p.name, str(p.id)))
        strategy, reason = "category_fallback", "Matches a category you explored; ML fallback"
    else:
        candidates.sort(key=lambda p: (not p.featured, p.name, str(p.id)))
    # Current DB stock/category state forms part of the cache version. Serving
    # never trusts cached availability after a checkout or external stock change.
    signature = [(str(p.id), str(p.updated_at), p.inventory_record.quantity) for p in products if p.inventory_record]
    generation = catalogue_version(redis_client) if redis_client is not None else "uncached"
    key = "recommendations:v1:" + sha256(json.dumps([generation, mode, user_id, sorted(map(str, seeds)),
                 sorted(map(str, purchased)), limit, signature, sorted((str(k), v) for k,v in counts.items())],
                 sort_keys=True).encode()).hexdigest()
    def compute():
        return RecommendationResponse(items=[RecommendationItem(product=_to_product_response(p),
                   score=1 / (index + 1), reason=reason) for index, p in enumerate(candidates[:limit])],
                   strategy=strategy, modelVersion="baseline-serving-v1").model_dump(mode="json")
    payload = cached_recommendations(redis_client, key, compute) if redis_client is not None else compute()
    response = RecommendationResponse.model_validate(payload)
    # Refresh product objects from authoritative rows even on a cache hit.
    response.items = [item.model_copy(update={"product": _to_product_response(available[item.product.id])})
                      for item in response.items if item.product.id in available
                      and item.product.id not in seeds and item.product.id not in purchased]
    return response
cached_recommendationsrecommendations/cache.py
def cached_recommendations(client, key: str, compute, ttl: int = 60):
    if type(ttl) is not int or ttl <= 0:
        raise ValueError("TTL must be a positive integer")
    try:
        encoded = client.get(key)
        if encoded is not None:
            return json.loads(encoded)
    except (RedisError, ValueError, TypeError, UnicodeDecodeError):
        pass
    result = compute()  # Model/database errors must propagate, unlike cache outages.
    try:
        client.setex(key, ttl, json.dumps(result))
    except RedisError:
        pass
    return result
